📖 Описание на системата

2L1P е интегрирана система за резервация на пътувания, която комбинира реални данни за полети и хотели с интелигентно управление на тарифите и нощувките. Системата поддържа двупосочни и еднопосочни полети, автоматично изчисление на нощувките при двупосочни полети и генериране на детайлни оферти в PDF, Word и Online формат.

            this.offerGenerated = true;
            const offer = {
                offerId: `2L1P-REAL-${Date.now()}`,
                timestamp: new Date().toLocaleString('bg-BG'),
                departingFlight: departingFlight,
                returningFlight: returningFlight,
                hotel: hotel,
                selectedTariff: tariff,
                hotelNights: this.hotelNights,
                clientEmail: email,
                currency: 'EUR'
            };
            let total = 0;
            if (tariff) {
                const tariffPrices = { ecofly: 125, extrafly: 155, flexfly: 187 };
                total += tariffPrices[tariff];
            }
            if (returningFlight && tariff) {
                const tariffPrices = { ecofly: 125, extrafly: 155, flexfly: 187 };
                total += tariffPrices[tariff];
            }
            if (hotel) {
                total += hotel.price * this.hotelNights;
            }
            offer.totalPrice = total;
            return offer;
        },

        // 🏨 ИЗЧИСЛЯВАНЕ НА НОЩУВКИ
        calculateHotelNights(departDate, returnDate) {
            if (!departDate || !returnDate) return 3;
            const checkin = new Date(departDate);
            const checkout = new Date(returnDate);
            const timeDiff = checkout.getTime() - checkin.getTime();
            const nights = Math.ceil(timeDiff / (1000 * 3600 * 24));
            return Math.max(nights, 1);
        }
    };
